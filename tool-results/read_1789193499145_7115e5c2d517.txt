   353→  async function stopRecording() {
   354→    if (!recorderRef.current || !activePrompt) return;
   355→    setRecording(false);
   356→    const heldMs = Date.now() - (recordStartRef.current || 0);
   357→    // v3.0 fix — ASR providers reject clips below their minimum duration;
   358→    // a too-short tap must produce supportive feedback, never a NaN score.
   359→    if (heldMs < 700) {
   360→      recorderRef.current.stop();
   361→      playIncorrect();
   362→      setResult({
   363→        verdict: "retry", accuracy: 0,
   364→        message: lang === "fr"
   365→          ? "Trop court ! Maintiens le bouton et dis toute la phrase. 🎤"
   366→          : "Too short! Hold the button and say the whole phrase. 🎤",
   367→        transcription: "—",
   368→      });
   369→      return;
   370→    }
   371→    setEvaluating(true);
   372→    const { wavBase64 } = recorderRef.current.stop();
   373→    // v3.0 §5.1 — ASR attempts (voice interaction tracking)
   374→    trackEvent("asr_attempt", { lessonId: plan?.lesson_id, lang: activePrompt?.isGf ? voiceLang : lang === "fr" ? "fr" : "en" });
   375→    try {
   376→      const res = await fetch("/api/pronunciation", {
   377→        method: "POST",
   378→        headers: { "Content-Type": "application/json" },
   379→        body: JSON.stringify({ audioBase64: wavBase64, target: activePrompt.target, lang: activePrompt.isGf ? voiceLang : lang === "fr" ? "fr" : "en" }),
   380→      });
   381→      if (!res.ok) throw new Error("evaluation unavailable");
   382→      const data = await res.json();
   383→      const accuracy = typeof data.accuracy === "number" && Number.isFinite(data.accuracy) ? data.accuracy : 0;
   384→      const r = {
   385→        verdict: data.verdict as string,
   386→        accuracy,
   387→        message: lang === "fr" ? data.messageFr : data.message,
   388→        transcription: data.transcription || "",
   389→        toneAccuracy: (data.tone_accuracy ?? null) as number | null,
   390→        toneAware: !!data.tone_aware,
   391→      };
   392→      setResult(r);
   393→      setScores((s) => [...s, r.accuracy]);
   394→      if (r.verdict === "correct") { playCorrect(); setStars((v) => v + 3); }
   395→      else if (r.verdict === "close") { playXp(); setStars((v) => v + 2); }
   396→      else playIncorrect();
   397→    } catch {
   398→      // Offline / API failure → gentle supportive feedback (never punitive)
   399→      setResult({
   400→        verdict: "close", accuracy: 60,
   401→        message: lang === "fr" ? "Je n'ai pas pu évaluer maintenant. Essaie encore !" : "I could not evaluate just now. Try again!",
   402→        transcription: "",
   403→      });
   404→      playIncorrect();
   405→    } finally {
   406→      setEvaluating(false);
   407→    }
   408→  }
