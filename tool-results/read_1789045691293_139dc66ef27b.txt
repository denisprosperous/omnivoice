     1→"use client";
     2→// ============================================================================
     3→// LESSON PLAYER — 5-phase bite-sized voice-first lesson (Master Prompt v2.0 §5.3)
     4→// Voice Hook (multilingual: en/fr/bkm/lns/byv) → Listen & Learn → Speak &
     5→// Practice (tone-aware ASR eval for Grassfields languages) → Apply & Create
     6→// (STS role-play, code-switching welcome) → Celebrate (badge + makossa).
     7→// Gamification per phase. Grassfields language packs surfaced offline.
     8→// v3.0: Bayangi (byv) wired end-to-end with Directive 9 placeholder states —
     9→// undocumented content renders a data-collection notice, never fake audio.
    10→// ============================================================================
    11→import React from "react";
    12→import { useApp } from "@/lib/store";
    13→import { t } from "@/lib/i18n";
    14→import { PatternBand, CharacterBubble, MicButton, Spinner, StatPill } from "./shared";
    15→import { getCharacter } from "@/lib/characters";
    16→import { VOICE_LANGUAGES, isGrassfields, isPlaceholderPhrase } from "@/lib/data/grassfields";
    17→import { WavRecorder, speak, playWavBase64, transcribe, speakFallback } from "@/lib/voice-client";
    18→import { trackEvent } from "@/lib/analytics";
    19→import { playCorrect, playIncorrect, playBadge, playCelebration, playLevelUp, playXp, startAmbient, stopAmbient } from "@/lib/sound-engine";
    20→import { Button } from "@/components/ui/button";
    21→import { cn } from "@/lib/utils";
    22→
    23→interface PracticePrompt { prompt: string; promptFr?: string; promptBkm?: string; promptLns?: string; promptByv?: string; target: string; targetBkm?: string; targetLns?: string; targetByv?: string; evaluation: string; kind?: string }
    24→interface LessonPlanData {
    25→  lesson_id: string; subject: string; level: string; isced_level: number;
    26→  integrated_learning_theme: string; sub_theme: string; week: number; month: number;
    27→  cefr_alignment: string; ib_learner_profile: string[]; supported_languages?: string[];
    28→  expected_learning_outcomes: string[]; teaching_strategies: string[];
    29→  didactic_materials: { physical: string[]; digital: string[] };
    30→  voice_assets: {
    31→    hook: { character: string; text: string; textFr: string; textBkm?: string; textLns?: string; textByv?: string; languages: string[] };
    32→    instruction: { text: string; textFr: string; textBkm?: string; textLns?: string; textByv?: string };
    33→    learn_content: { title: string; titleFr: string; lines: string[]; linesFr: string[]; visual: string };
    34→    practice_prompts: PracticePrompt[];
    35→    feedback: { correct: string; incorrect: string; encouragement: string };
    36→    celebration: { character: string; text: string; textFr: string };
    37→    background_music: string;
    38→  };
    39→  voice_interactions: Array<{ type: string; prompt: string; asr_target?: string; evaluation_criteria: string[] }>;
    40→  sts_scenario?: { description: string; descriptionFr: string; character: string; opener: string };
    41→  gamification: {
    42→    mechanics: string[]; xp_points: number; badge_name: string; badge_code: string;
    43→    voice_challenge: { description: string; descriptionFr: string; descriptionBkm?: string; descriptionLns?: string; descriptionByv?: string; evaluation: string };
    44→  };
    45→  activities: Array<{ phase: string; duration: string; description: string; descriptionFr: string }>;
    46→  assessment: { criteria: string[]; methods: string[] };
    47→  offline_capability: { downloadable: boolean; size_mb: number; components: string[]; grassfields_language_packs?: { bkm: string; lns: string } };
    48→  differentiation: string[];
    49→  cultural_notes: string; cultural_notesFr: string;
    50→  native_speaker_review?: string;
    51→}
    52→const PHASES = ["hook", "listen", "speak", "apply", "celebrate"] as const;
    53→type Phase = (typeof PHASES)[number];
    54→
    55→export function LessonPlayer() {
    56→  const { learner, lang, voiceLang, currentLessonId, setView } = useApp();
    57→  const [plan, setPlan] = React.useState<LessonPlanData | null>(null);
    58→  const [loading, setLoading] = React.useState(true);
    59→  const [phase, setPhase] = React.useState<Phase>("hook");
    60→  const [hookDone, setHookDone] = React.useState(false);
    61→  // Hook listening language (§7.3: learner may hear the hook in en/fr/bkm/lns/byv)
    62→  const [hookLang, setHookLang] = React.useState<string>("en");
    63→
    64→  // Practice state
    65→  const [promptIdx, setPromptIdx] = React.useState(0);
    66→  const [recording, setRecording] = React.useState(false);
    67→  const [evaluating, setEvaluating] = React.useState(false);
    68→  const [result, setResult] = React.useState<{ verdict: string; accuracy: number; message: string; transcription: string; toneAccuracy?: number | null; toneAware?: boolean } | null>(null);
    69→  const [stars, setStars] = React.useState(0);
    70→  const [scores, setScores] = React.useState<number[]>([]);
    71→  const recorderRef = React.useRef<WavRecorder | null>(null);
    72→  const recordStartRef = React.useRef<number>(0);
    73→
    74→  // Apply (STS) state
    75→  const [stsHistory, setStsHistory] = React.useState<Array<{ role: "user" | "assistant"; content: string }>>([]);
    76→  const [stsBusy, setStsBusy] = React.useState(false);
    77→  const [stsOpened, setStsOpened] = React.useState(false);
    78→
    79→  // Celebrate state
    80→  const [awarded, setAwarded] = React.useState<{ xp: number; badge: string | null; levelUp: { level: number; title: string } | null } | null>(null);
    81→
    82→  React.useEffect(() => {
    83→    if (!currentLessonId) return;
    84→    setLoading(true);
    85→    fetch(`/api/lessons?id=${currentLessonId}`)
    86→      .then((r) => r.json())
    87→      .then((d) => { setPlan(d.lesson?.plan || null); })
    88→      .finally(() => setLoading(false));
    89→  }, [currentLessonId]);
    90→
    91→  // Default the hook language to the learner's chosen voice language when available (§7.3)
    92→  React.useEffect(() => {
    93→    if (!plan) return;
    94→    const h = plan.voice_assets.hook;
    95→    const available = (h.languages || ["en", "fr"]).filter((l) =>
    96→      l === "bkm" ? !!h.textBkm : l === "lns" ? !!h.textLns : l === "byv" ? !!h.textByv : true
    97→    );
    98→    setHookLang(available.includes(voiceLang) ? voiceLang : "en");
    99→  }, [plan, voiceLang]);
   100→
   101→  // Ambient background music per ILT mood (stipulated per-ILT BGM)
   102→  React.useEffect(() => {
   103→    startAmbient("home"); // Month 1 = The Home mood
   104→    return () => stopAmbient();
   105→  }, []);
   106→
   107→  async function finishLesson() {
   108→    if (!learner || !plan) return;
   109→    const total = scores.length ? Math.round(scores.reduce((a, b) => a + b, 0) / scores.length) : 80;
   110→    // v3.0 §5.1 — lesson completion rate (pedagogical effectiveness metric)
   111→    trackEvent("lesson_completion", { lessonId: plan.lesson_id, voiceLang, avgScore: total });
   112→    try {
   113→      const res = await fetch("/api/learner", {
   114→        method: "PUT",
   115→        headers: { "Content-Type": "application/json" },
   116→        body: JSON.stringify({
   117→          learnerId: learner.id,
   118→          xp: plan.gamification.xp_points,
   119→          reason: `Lesson: ${plan.lesson_id}`,
   120→          badgeCode: plan.gamification.badge_code,
   121→          skillCode: undefined,
   122→          assessment: {
   123→            lessonId: plan.lesson_id,
   124→            type: "formative",
   125→            score: total,
   126→            details: { scores, stars },
   127→          },
   128→        }),
   129→      });
   130→      const data = await res.json();
   131→      setAwarded({
   132→        xp: plan.gamification.xp_points,
   133→        badge: data.badge ? plan.gamification.badge_name : null,
   134→        levelUp: data.levelUp ? { level: data.levelUp.level, title: data.levelUp.title } : null,
   135→      });
   136→    } catch {
   137→      setAwarded({ xp: plan.gamification.xp_points, badge: null, levelUp: null });
   138→    }
   139→  }
   140→
   141→  if (loading) return <div className="min-h-screen bg-[#FFFBEB]"><PatternBand /><Spinner label={lang === "fr" ? "Chargement de la quête..." : "Loading quest..."} /></div>;
   142→  if (!plan) {
   143→    return (
   144→      <div className="min-h-screen bg-[#FFFBEB]">
   145→        <PatternBand />
   146→        <div className="p-8 text-center text-amber-800">{lang === "fr" ? "Leçon introuvable." : "Lesson not found."}</div>
   147→      </div>
   148→    );
   149→  }
   150→
   151→  const va = plan.voice_assets;
   152→  const shown = (en: string, fr: string) => (lang === "fr" ? fr : en);
   153→  const prompts = va.practice_prompts;
   154→  // §7.3 — resolve the practice prompt in the learner's voice language
   155→  const promptForVoice = (pp: PracticePrompt): { prompt: string; target: string; isGf: boolean } => {
   156→    if (voiceLang === "bkm" && pp.promptBkm) return { prompt: pp.promptBkm, target: pp.targetBkm || pp.target, isGf: true };
   157→    if (voiceLang === "lns" && pp.promptLns) return { prompt: pp.promptLns, target: pp.targetLns || pp.target, isGf: true };
   158→    if (voiceLang === "byv" && pp.promptByv) return { prompt: pp.promptByv, target: pp.targetByv || pp.target, isGf: true };
   159→    return { prompt: lang === "fr" ? pp.promptFr || pp.prompt : pp.prompt, target: pp.target, isGf: false };
   160→  };
   161→  const currentPrompt = prompts[promptIdx];
   162→  const activePrompt = currentPrompt ? promptForVoice(currentPrompt) : null;
   163→  const char = getCharacter(va.hook.character);
   164→  // §7.3 multilingual hook — hook text in the selected listening language
   165→  const hookText =
   166→    hookLang === "fr" ? va.hook.textFr
   167→    : hookLang === "bkm" ? (va.hook.textBkm || va.hook.text)
   168→    : hookLang === "lns" ? (va.hook.textLns || va.hook.text)
   169→    : hookLang === "byv" ? (va.hook.textByv || va.hook.text)
   170→    : va.hook.text;
   171→  const hookIsPlaceholder = isPlaceholderPhrase(hookText);
   172→  const hookLanguages = (va.hook.languages || ["en", "fr"]).filter((l) =>
   173→    l === "bkm" ? !!va.hook.textBkm : l === "lns" ? !!va.hook.textLns : l === "byv" ? !!va.hook.textByv : l === "ewo" ? true : true
   174→  );
   175→
   176→  // ---------- Practice handlers ----------
   177→  async function startRecording() {
   178→    setResult(null);
   179→    recordStartRef.current = Date.now();
   180→    try {
   181→      recorderRef.current = new WavRecorder();
   182→      await recorderRef.current.start();
   183→      setRecording(true);
   184→    } catch {
   185→      playIncorrect();
   186→      setResult({
   187→        verdict: "retry", accuracy: 0,
   188→        message: lang === "fr"
   189→          ? "Je ne peux pas accéder au micro. Demande à un adulte d'autoriser le microphone, puis réessaie !"
   190→          : "I cannot reach the microphone. Ask a grown-up to allow the microphone, then try again!",
   191→        transcription: "—",
   192→      });
   193→    }
   194→  }
   195→
   196→  async function stopRecording() {
   197→    if (!recorderRef.current || !activePrompt) return;
   198→    setRecording(false);
   199→    const heldMs = Date.now() - (recordStartRef.current || 0);
   200→    // v3.0 fix — ASR providers reject clips below their minimum duration;
   201→    // a too-short tap must produce supportive feedback, never a NaN score.
   202→    if (heldMs < 700) {
   203→      recorderRef.current.stop();
   204→      playIncorrect();
   205→      setResult({
   206→        verdict: "retry", accuracy: 0,
   207→        message: lang === "fr"
   208→          ? "Trop court ! Maintiens le bouton et dis toute la phrase. 🎤"
   209→          : "Too short! Hold the button and say the whole phrase. 🎤",
   210→        transcription: "—",
   211→      });
   212→      return;
   213→    }
   214→    setEvaluating(true);
   215→    const { wavBase64 } = recorderRef.current.stop();
   216→    // v3.0 §5.1 — ASR attempts (voice interaction tracking)
   217→    trackEvent("asr_attempt", { lessonId: plan?.lesson_id, lang: activePrompt?.isGf ? voiceLang : lang === "fr" ? "fr" : "en" });
   218→    try {
   219→      const res = await fetch("/api/pronunciation", {
   220→        method: "POST",
   221→        headers: { "Content-Type": "application/json" },
   222→        body: JSON.stringify({ audioBase64: wavBase64, target: activePrompt.target, lang: activePrompt.isGf ? voiceLang : lang === "fr" ? "fr" : "en" }),
   223→      });
   224→      if (!res.ok) throw new Error("evaluation unavailable");
   225→      const data = await res.json();
   226→      const accuracy = typeof data.accuracy === "number" && Number.isFinite(data.accuracy) ? data.accuracy : 0;
   227→      const r = {
   228→        verdict: data.verdict as string,
   229→        accuracy,
   230→        message: lang === "fr" ? data.messageFr : data.message,
   231→        transcription: data.transcription || "",
   232→        toneAccuracy: (data.tone_accuracy ?? null) as number | null,
   233→        toneAware: !!data.tone_aware,
   234→      };
   235→      setResult(r);
   236→      setScores((s) => [...s, r.accuracy]);
   237→      if (r.verdict === "correct") { playCorrect(); setStars((v) => v + 3); }
   238→      else if (r.verdict === "close") { playXp(); setStars((v) => v + 2); }
   239→      else playIncorrect();
   240→    } catch {
   241→      // Offline / API failure → gentle supportive feedback (never punitive)
   242→      setResult({
   243→        verdict: "close", accuracy: 60,
   244→        message: lang === "fr" ? "Je n'ai pas pu évaluer maintenant. Essaie encore !" : "I could not evaluate just now. Try again!",
   245→        transcription: "",
   246→      });
   247→      playIncorrect();
   248→    } finally {
   249→      setEvaluating(false);
   250→    }
   251→  }
   252→
   253→  async function nextPrompt() {
   254→    if (promptIdx < prompts.length - 1) {
   255→      setPromptIdx((i) => i + 1);
   256→      setResult(null);
   257→    } else {
   258→      setPhase("apply");
   259→    }
   260→  }
   261→
   262→  // ---------- STS handlers ----------
   263→  async function openSts() {
   264→    if (!plan.sts_scenario || stsOpened) return;
   265→    setStsOpened(true);
   266→    setStsBusy(true);
   267→    const opener = plan.sts_scenario.opener;
   268→    setStsHistory([{ role: "assistant", content: opener }]);
   269→    void speak(opener, plan.sts_scenario.character, lang);
   270→    setTimeout(() => setStsBusy(false), 2500);
   271→  }
   272→
   273→  async function stsTurn() {
   274→    if (!plan.sts_scenario || stsBusy) return;
   275→    setStsBusy(true);
   276→    try {
   277→      recorderRef.current = new WavRecorder();
   278→      await recorderRef.current.start();
   279→      setRecording(true);
   280→    } catch {
   281→      setStsBusy(false);
   282→    }
   283→  }
   284→
   285→  async function stsStop() {
   286→    if (!recorderRef.current || !plan.sts_scenario) return;
   287→    setRecording(false);
   288→    setStsBusy(true);
   289→    const { wavBase64 } = recorderRef.current.stop();
   290→    try {
   291→      const res = await fetch("/api/sts", {
   292→        method: "POST",
   293→        headers: { "Content-Type": "application/json" },
   294→        body: JSON.stringify({
   295→          audioBase64: wavBase64,
   296→          character: plan.sts_scenario.character,
   297→          scenario: plan.sts_scenario.description,
   298→          history: stsHistory,
   299→          learnerName: learner?.name || "my friend",
   300→          learnerLevel: plan.level,
   301→          lang: isGrassfields(voiceLang) ? voiceLang : lang === "fr" ? "fr" : "en",
   302→        }),
   303→      });
   304→      const data = await res.json();
   305→      if (data.reply) {
   306→        setStsHistory((h) => [
   307→          ...h,
   308→          { role: "user", content: data.transcript || "…" },
   309→          { role: "assistant", content: data.reply },
   310→        ]);
   311→        if (data.audioBase64) playWavBase64(data.audioBase64, data.pitchRate || 1);
   312→        else void speak(data.reply, plan.sts_scenario.character, lang);
   313→        playXp();
   314→      } else if (data.error === "no_speech") {
   315→        const msg = lang === "fr" ? "Je ne t'ai pas entendu — essaie encore !" : "I did not hear you — try again!";
   316→        setStsHistory((h) => [...h, { role: "assistant", content: msg }]);
   317→        void speak(msg, plan.sts_scenario.character, lang);
   318→      }
   319→    } catch {
   320→      const msg = lang === "fr" ? "Hmm, réessayons dans un instant." : "Hmm, let us try again in a moment.";
   321→      setStsHistory((h) => [...h, { role: "assistant", content: msg }]);
   322→    } finally {
   323→      setStsBusy(false);
   324→    }
   325→  }
   326→
   327→  const phaseMeta: Record<Phase, { label: string; icon: string }> = {
   328→    hook: { label: t("hook", lang), icon: "🎣" },
   329→    listen: { label: t("listen", lang), icon: "👂🏾" },
   330→    speak: { label: t("speak", lang), icon: "🎤" },
   331→    apply: { label: t("apply", lang), icon: "✨" },
   332→    celebrate: { label: t("celebrate", lang), icon: "🎉" },
   333→  };
   334→
   335→  return (
   336→    <main className="min-h-screen bg-[#FFFBEB] pb-28">
   337→      <PatternBand />
   338→      {/* Progress header */}
   339→      <div className="sticky top-[52px] z-30 border-b border-amber-200 bg-[#FFFBEB]/95 backdrop-blur">
   340→        <div className="mx-auto flex max-w-3xl items-center gap-1.5 px-3 py-2">
   341→          {PHASES.map((p, i) => (
   342→            <React.Fragment key={p}>
   343→              <button
   344→                onClick={() => { if (PHASES.indexOf(phase) > i) setPhase(p); }}
   345→                className={cn(
   346→                  "flex min-h-[34px] items-center gap-1 rounded-full border-2 px-2.5 text-[11px] font-bold transition-all sm:text-xs",
   347→                  phase === p ? "border-amber-600 bg-amber-600 text-white shadow" : "border-amber-200 bg-white text-amber-700"
   348→                )}
   349→                aria-current={phase === p ? "step" : undefined}
   350→              >
   351→                <span aria-hidden>{phaseMeta[p].icon}</span>
   352→                <span className="hidden sm:inline">{phaseMeta[p].label}</span>
   353→              </button>
   354→              {i < PHASES.length - 1 && <span className="h-0.5 flex-1 bg-amber-200" aria-hidden />}
   355→            </React.Fragment>
   356→          ))}
   357→        </div>
   358→      </div>
   359→
   360→      <div className="mx-auto max-w-3xl space-y-4 px-4 pt-4">
   361→        {/* Lesson meta */}
   362→        <div className="flex flex-wrap items-center gap-2">
   363→          <StatPill icon="📚" label={plan.subject} value={plan.level} />
   364→          <StatPill icon="🌍" label="CEFR" value={plan.cefr_alignment} color="#065F46" />
   365→          <StatPill icon="🎯" label="XP" value={`+${plan.gamification.xp_points}`} color="#9D174D" />
   366→          {plan.gamification.mechanics.map((m) => (
   367→            <StatPill key={m} icon="🎮" label="mechanic" value={m} color="#7C2D12" />
   368→          ))}
   369→        </div>
   370→
   371→        {/* PHASE 1 — Voice Hook (multilingual per §7.3) */}
   372→        {phase === "hook" && (
   373→          <section aria-label={phaseMeta.hook.label} className="space-y-4">
   374→            <h1 className="text-xl font-extrabold text-amber-900 sm:text-2xl">
   375→              {shown(va.learn_content.title, va.learn_content.titleFr)}
   376→            </h1>
   377→            {/* Hook language switcher — listen in en/fr/bkm/lns (§7.3 voice hook) */}
   378→            {hookLanguages.length > 1 && (
   379→              <div className="flex flex-wrap items-center gap-1.5 rounded-xl border-2 border-lime-200 bg-lime-50/60 p-2">
   380→                <span className="px-1 text-[11px] font-bold uppercase tracking-wide text-lime-800">🪶 {t("hookLanguage", lang)}:</span>
   381→                {hookLanguages.map((l) => {
   382→                  const vl = VOICE_LANGUAGES.find((v) => v.id === l);
   383→                  return (
   384→                    <button
   385→                      key={l}
   386→                      onClick={() => setHookLang(l)}
   387→                      aria-pressed={hookLang === l}
   388→                      className={cn(
   389→                        "min-h-[32px] rounded-full border-2 px-2.5 text-[11px] font-bold transition-all",
   390→                        hookLang === l ? "border-lime-700 bg-lime-700 text-white" : "border-lime-300 bg-white text-lime-800 hover:border-lime-500"
   391→                      )}
   392→                    >
   393→                      {vl ? vl.label : l}
   394→                    </button>
   395→                  );
   396→                })}
   397→              </div>
   398→            )}
   399→            {hookIsPlaceholder ? (
   400→              // v3.0 Directive 9 gate — Bayangi placeholder: documented-by-natives notice,
   401→              // never synthesized placeholder audio.
   402→              <div className="rounded-2xl border-2 border-dashed border-lime-400 bg-lime-50/80 p-4" role="status">
   403→                <p className="text-sm font-extrabold text-lime-900">🪶 Bayangi (Banyangi) — {t("pendingDocumentation", lang)}</p>
   404→                <p className="mt-1 text-xs leading-relaxed text-lime-800">{hookText}</p>
   405→                <p className="mt-2 text-[11px] font-semibold text-lime-700">
   406→                  ⏳ {t("dataCollection", lang)}: 500h · SIL Cameroon / Local Community · Q2 2025
   407→                </p>
   408→              </div>
   409→            ) : (
   410→              <CharacterBubble
   411→                characterId={va.hook.character}
   412→                text={hookText}
   413→                textFr={hookText}
   414→                lang={hookLang === "fr" ? "fr" : "en"}
   415→                speakLang={hookLang}
   416→                onSpeakDone={() => setHookDone(true)}
   417→              />
   418→            )}
   419→            {isGrassfields(hookLang) && (
   420→              <p className="rounded-xl bg-lime-50 px-3 py-2 text-[11px] font-semibold text-lime-900">
   421→                🪶 GACL tone-marked {VOICE_LANGUAGES.find((v) => v.id === hookLang)?.label || hookLang} · {t("codeSwitch", lang)}
   422→                {hookIsPlaceholder && <span className="ml-1">· ⏳ {t("pendingDocumentation", lang)}</span>}
   423→                {!hookIsPlaceholder && plan.native_speaker_review === "validated" && <span className="ml-1">· ✅ {t("nativeReview", lang)}</span>}
   424→              </p>
   425→            )}
   426→            <div className="rounded-2xl border-2 border-amber-200 bg-white p-4">
   427→              <h3 className="mb-1 text-sm font-bold text-amber-900">📖 {lang === "fr" ? "Histoire du thème" : "Theme story"}</h3>

   713→/** Standalone voice challenge recorder (for lessons without STS scenario) */
   714→function VoiceChallengeRecorder({ onDone, lang }: { onDone: () => void; lang: string }) {
   715→  const [recording, setRecording] = React.useState(false);
   716→  const [saved, setSaved] = React.useState(false);
   717→  const recRef = React.useRef<WavRecorder | null>(null);
   718→
   719→  async function down() {
   720→    recRef.current = new WavRecorder();
   721→    try {
   722→      await recRef.current.start();
   723→      setRecording(true);
   724→    } catch { /* mic unavailable */ }
   725→  }
   726→  function up() {
   727→    if (!recRef.current) return;
   728→    setRecording(false);
   729→    recRef.current.stop(); // demo capture — human-in-the-loop portfolio piece
   730→    setSaved(true);
   731→    playBadge();
   732→  }
   733→  return (
   734→    <div className="flex flex-col items-center gap-3">
   735→      <MicButton recording={recording} onDown={down} onUp={up} label={t("record", lang)} />
   736→      {saved && (
   737→        <Button className="h-12 bg-lime-700 px-8 font-extrabold hover:bg-lime-800" onClick={onDone}>
   738→          ✓ {t("complete", lang)}
   739→        </Button>
   740→      )}
   741→    </div>
   742→  );
   743→}
