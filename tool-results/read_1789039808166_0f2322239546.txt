   563→   563→/** Standalone voice challenge recorder (for lessons without STS scenario) */
   564→   564→function VoiceChallengeRecorder({ onDone, lang }: { onDone: () => void; lang: string }) {
   565→   565→  const [recording, setRecording] = React.useState(false);
   566→   566→  const [saved, setSaved] = React.useState(false);
   567→   567→  const recRef = React.useRef<WavRecorder | null>(null);
   568→   568→
   569→   569→  async function down() {
   570→   570→    recRef.current = new WavRecorder();
   571→   571→    try {
   572→   572→      await recRef.current.start();
   573→   573→      setRecording(true);
   574→   574→    } catch { /* mic unavailable */ }
   575→   575→  }
   576→   576→  function up() {
   577→   577→    if (!recRef.current) return;
   578→   578→    setRecording(false);
   579→   579→    recRef.current.stop(); // demo capture — human-in-the-loop portfolio piece
   580→   580→    setSaved(true);
   581→   581→    playBadge();
   582→   582→  }
   583→   583→  return (
   584→   584→    <div className="flex flex-col items-center gap-3">
   585→   585→      <MicButton recording={recording} onDown={down} onUp={up} label={t("record", lang)} />
   586→   586→      {saved && (
   587→   587→        <Button className="h-12 bg-lime-700 px-8 font-extrabold hover:bg-lime-800" onClick={onDone}>
   588→   588→          ✓ {t("complete", lang)}
   589→   589→        </Button>
   590→   590→      )}
   591→   591→    </div>
   592→   592→  );
   593→   593→}
