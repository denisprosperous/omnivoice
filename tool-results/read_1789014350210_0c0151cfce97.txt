   584→   584→### 5.5 Voice Pipeline Implementation
   585→   585→
   586→   586→```python
   587→   587→# Complete voice pipeline for STS
   588→   588→class VoicePipeline:
   589→   589→    def __init__(self):
   590→   590→        self.asr = load_model("UBC-NLP/Simba-S")  # African ASR
   591→   591→        self.tts = EdgeTTS()  # Free TTS
   592→   592→        self.llm = OpenAIClient()  # Content generation
   593→   593→    
   594→   594→    async def speech_to_speech(self, audio_input: bytes) -> bytes:
   595→   595→        # Step 1: Speech to Text
   596→   596→        text = await self.asr.transcribe(audio_input)
   597→   597→        
   598→   598→        # Step 2: Generate Response
   599→   599→        response_text = await self.llm.generate(
   600→   600→            prompt=text,
   601→   601→            context=self.get_lesson_context()
   602→   602→        )
   603→   603→        
   604→   604→        # Step 3: Text to Speech
   605→   605→        audio_output = await self.tts.synthesize(
   606→   606→            text=response_text,
   607→   607→            voice="cameroonian_child_friendly"
   608→   608→        )
   609→   609→        
   610→   610→        return audio_output
   611→   611→    
   612→   612→    async def evaluate_pronunciation(
   613→   613→        self, 
   614→   614→        learner_audio: bytes, 
   615→   615→        target_text: str
   616→   616→    ) -> dict:
   617→   617→        # Transcribe learner speech
   618→   618→        transcription = await self.asr.transcribe(learner_audio)
   619→   619→        
   620→   620→        # Compare with target
   621→   621→        similarity = calculate_similarity(transcription, target_text)
   622→   622→        
   623→   623→        return {
   624→   624→            "transcription": transcription,
   625→   625→            "target": target_text,
   626→   626→            "accuracy": similarity,
   627→   627→            "feedback": generate_feedback(similarity)
   628→   628→        }
   629→   629→```
