    92→    92→
    93→    93→**Integration with Platform:**
    94→    94→```python
    95→    95→# server/services/kokoro_tts.py
    96→    96→from kokoro import KPipeline
    97→    97→import io
    98→    98→import soundfile as sf
    99→    99→
   100→   100→class KokoroTTSService:
   101→   101→    def __init__(self):
   102→   102→        self.pipelines = {
   103→   103→            'en': KPipeline(lang_code='a'),  # American English
   104→   104→            'fr': KPipeline(lang_code='f'),  # French
   105→   105→        }
   106→   106→        self.default_voices = {
   107→   107→            'en': 'af_bella',
   108→   108→            'fr': 'ff_siwis',
   109→   109→        }
   110→   110→    
   111→   111→    async def synthesize(
   112→   112→        self,
   113→   113→        text: str,
   114→   114→        language: str = 'en',
   115→   115→        voice: str = None,
   116→   116→        speed: float = 1.0
   117→   117→    ) -> bytes:
   118→   118→        """Generate natural-sounding speech."""
   119→   119→        pipeline = self.pipelines.get(language, self.pipelines['en'])
   120→   120→        voice = voice or self.default_voices.get(language, 'af_bella')
   121→   121→        
   122→   122→        # Generate audio
   123→   123→        generator = pipeline(text, voice=voice, speed=speed)
   124→   124→        
   125→   125→        # Collect audio chunks
   126→   126→        audio_chunks = []
   127→   127→        for _, _, audio in generator:
   128→   128→            audio_chunks.append(audio)
   129→   129→        
   130→   130→        # Concatenate and convert to bytes
   131→   131→        import numpy as np
   132→   132→        full_audio = np.concatenate(audio_chunks)
   133→   133→        
   134→   134→        buffer = io.BytesIO()
   135→   135→        sf.write(buffer, full_audio, 24000, format='WAV')
   136→   136→        buffer.seek(0)
   137→   137→        
   138→   138→        return buffer.read()
   139→   139→```

   178→   178→

   190→   190→**Integration with Platform:**
   191→   191→```python
   192→   192→# server/services/faster_whisper_stt.py
   193→   193→from faster_whisper import WhisperModel
   194→   194→
   195→   195→class FasterWhisperSTTService:
   196→   196→    def __init__(self):
   197→   197→        # Load models for different languages
   198→   198→        self.models = {
   199→   199→            'base': WhisperModel("base", device="cuda", compute_type="float16"),
   200→   200→            'small': WhisperModel("small", device="cuda", compute_type="float16"),
   201→   201→        }
   202→   202→    
   203→   203→    async def transcribe(
   204→   204→        self,
   205→   205→        audio_bytes: bytes,
   206→   206→        language: str = None,
   207→   207→        model_size: str = 'base'
   208→   208→    ) -> dict:
   209→   209→        """Transcribe audio to text with language detection."""
   210→   210→        # Save to temporary file
   211→   211→        import tempfile
   212→   212→        with tempfile.NamedTemporaryFile(suffix='.wav', delete=False) as f:
   213→   213→            f.write(audio_bytes)
   214→   214→            audio_path = f.name
   215→   215→        
   216→   216→        # Transcribe
   217→   217→        model = self.models.get(model_size, self.models['base'])
   218→   218→        segments, info = model.transcribe(
   219→   219→            audio_path,
   220→   220→            language=language,  # None for auto-detect
   221→   221→            beam_size=5,
   222→   222→            word_timestamps=True
   223→   223→        )
   224→   224→        
   225→   225→        # Collect results
   226→   226→        transcription = " ".join([segment.text for segment in segments])
   227→   227→        
   228→   228→        return {
   229→   229→            "text": transcription.strip(),
   230→   230→            "language": info.language,
   231→   231→            "language_probability": info.language_probability,
   232→   232→            "duration": info.duration
   233→   233→        }
   234→   234→```

   305→   305→#### Option B: MOSS-Speech (True Speech-to-Speech Without Text Guidance)
   306→   306→
   307→   307→MOSS-Speech introduces true end-to-end speech interaction. Unlike cascaded pipelines or text-guided models, it directly generates speech without first producing text. This overcomes the limitation of generated speech being constrained by a text bottleneck, enabling more natural and efficient speech-to-speech dialogue.
   308→   308→
   309→   309→**Key Features:**
   310→   310→- **True Speech-to-Speech Modeling:** No text guidance required
   311→   311→- **Layer-Splitting Architecture:** Integrates modality-specific layers on top of pretrained text LLM backbones
   312→   312→- **Frozen Pre-Training Strategy:** Preserves LLM reasoning while enhancing speech understanding and generation
   313→   313→- **Expressive & Efficient:** Maintains paralinguistic cues often lost in cascaded pipelines, such as tone, emotion, and prosody
   314→   314→
   315→   315→**Installation:**
   316→   316→```bash
   317→   317→# Clone repository
   318→   318→git clone https://github.com/OpenMOSS/MOSS-Speech
   319→   319→cd MOSS-Speech
   320→   320→
   321→   321→# Install dependencies
   322→   322→pip install -r requirements.txt
   323→   323→git submodule update --init --recursive
   324→   324→
   325→   325→# Launch web demo
   326→   326→python3 gradio_demo.py
   327→   327→```
   328→   328→
   329→   329→**Python API:**
   330→   330→```python
   331→   331→import os
   332→   332→from dataclasses import astuple
   333→   333→import torch
   334→   334→import torchaudio
   335→   335→from transformers import AutoModel, AutoProcessor, GenerationConfig
   336→   336→
   337→   337→model_path = "fnlp/MOSS-Speech"
   338→   338→codec_path = "fnlp/MOSS-Speech-Codec"
   339→   339→
   340→   340→processor = AutoProcessor.from_pretrained(
   341→   341→    model_path, 
   342→   342→    codec_path=codec_path, 
   343→   343→    device="cuda", 
   344→   344→    trust_remote_code=True
   345→   345→)
   346→   346→
   347→   347→model = AutoModel.from_pretrained(
   348→   348→    model_path, 
   349→   349→    trust_remote_code=True, 
   350→   350→    device_map="cuda"
   351→   351→).eval()
   352→   352→
   353→   353→messages = [
   354→   354→    {"role": "system", "content": "You are a helpful voice assistant for Cameroonian children."},
   355→   355→    {"role": "user", "content": "À bwɛ̀! Nà wù dà?"}
   356→   356→]
   357→   357→
   358→   358→encoded_inputs = processor(messages, "audio")
   359→   359→
   360→   360→with torch.inference_mode():
   361→   361→    token_ids = model.generate(
   362→   362→        input_ids=encoded_inputs["input_ids"].to("cuda"),
   363→   363→        attention_mask=encoded_inputs["attention_mask"].to("cuda"),
   364→   364→    )
   365→   365→
   366→   366→results = processor.decode(token_ids, "audio")
   367→   367→audio, text, sample_rate = astuple(results[0])
   368→   368→
   369→   369→# Save audio
   370→   370→torchaudio.save("response.wav", audio, sample_rate)
   371→   371→```

   574→   574→### 2.3 Unified Voice Service
   575→   575→
   576→   576→```python
   577→   577→# server/services/unified_voice.py
   578→   578→from typing import Optional
   579→   579→import numpy as np
   580→   580→
   581→   581→class UnifiedVoiceService:
   582→   582→    """Complete voice pipeline: STT → LLM → TTS with STS option."""
   583→   583→    
   584→   584→    def __init__(self):
   585→   585→        # Initialize STT
   586→   586→        from faster_whisper import WhisperModel
   587→   587→        self.stt_models = {
   588→   588→            "base": WhisperModel("base", device="cuda", compute_type="float16"),
   589→   589→            "small": WhisperModel("small", device="cuda", compute_type="float16"),
   590→   590→        }
   591→   591→        
   592→   592→        # Initialize TTS
   593→   593→        from kokoro import KPipeline
   594→   594→        self.tts_pipelines = {
   595→   595→            "en": KPipeline(lang_code='a'),
   596→   596→            "fr": KPipeline(lang_code='f'),
   597→   597→        }
   598→   598→        
   599→   599→        # Initialize STS (Chroma)
   600→   600→        self.chroma_url = "http://localhost:8000/v1/chat/completions"
   601→   601→    
   602→   602→    async def speech_to_text(
   603→   603→        self,
   604→   604→        audio_bytes: bytes,
   605→   605→        language: Optional[str] = None,
   606→   606→        model_size: str = "base"
   607→   607→    ) -> dict:
   608→   608→        """Convert speech to text."""
   609→   609→        import tempfile
   610→   610→        with tempfile.NamedTemporaryFile(suffix='.wav', delete=False) as f:
   611→   611→            f.write(audio_bytes)
   612→   612→            audio_path = f.name
   613→   613→        
   614→   614→        model = self.stt_models.get(model_size, self.stt_models["base"])
   615→   615→        segments, info = model.transcribe(
   616→   616→            audio_path,
   617→   617→            language=language,
   618→   618→            beam_size=5,
   619→   619→            word_timestamps=True
   620→   620→        )
   621→   621→        
   622→   622→        transcription = " ".join([seg.text for seg in segments])
   623→   623→        
   624→   624→        return {
   625→   625→            "text": transcription.strip(),
   626→   626→            "language": info.language,
   627→   627→            "confidence": info.language_probability,
   628→   628→        }
   629→   629→    
   630→   630→    async def text_to_speech(
   631→   631→        self,
   632→   632→        text: str,
   633→   633→        language: str = "en",
   634→   634→        voice: Optional[str] = None,
   635→   635→        speed: float = 1.0
   636→   636→    ) -> bytes:
   637→   637→        """Convert text to natural speech."""
   638→   638→        pipeline = self.tts_pipelines.get(language, self.tts_pipelines["en"])
   639→   639→        
   640→   640→        # Default voices per language
   641→   641→        default_voices = {
   642→   642→            "en": "af_bella",
   643→   643→            "fr": "ff_siwis",
   644→   644→        }
   645→   645→        voice = voice or default_voices.get(language, "af_bella")
   646→   646→        
   647→   647→        # Generate audio
   648→   648→        generator = pipeline(text, voice=voice, speed=speed)
   649→   649→        
   650→   650→        audio_chunks = []
   651→   651→        for _, _, audio in generator:
   652→   652→            audio_chunks.append(audio)
   653→   653→        
   654→   654→        full_audio = np.concatenate(audio_chunks)
   655→   655→        
   656→   656→        import io
   657→   657→        import soundfile as sf
   658→   658→        buffer = io.BytesIO()
   659→   659→        sf.write(buffer, full_audio, 24000, format='WAV')
   660→   660→        buffer.seek(0)
   661→   661→        
   662→   662→        return buffer.read()
   663→   663→    
   664→   664→    async def speech_to_speech(
   665→   665→        self,
   666→   666→        audio_bytes: bytes,
   667→   667→        language: str = "en",
   668→   668→        character: str = "kwe"
   669→   669→    ) -> bytes:
   670→   670→        """Complete speech-to-speech pipeline."""
   671→   671→        import requests
   672→   672→        import base64
   673→   673→        import tempfile
   674→   674→        
   675→   675→        # Save input audio
   676→   676→        with tempfile.NamedTemporaryFile(suffix='.wav', delete=False) as f:
   677→   677→            f.write(audio_bytes)
   678→   678→            audio_path = f.name
   679→   679→        
   680→   680→        # Call Chroma STS
   681→   681→        payload = {
   682→   682→            "model": "chroma",
   683→   683→            "messages": [
   684→   684→                {"role": "system", "content": self._get_character_prompt(character)},
   685→   685→                {"role": "user", "content": [{"type": "audio", "audio": audio_path}]}
   686→   686→            ],
   687→   687→            "max_tokens": 500,
   688→   688→            "return_audio": True
   689→   689→        }
   690→   690→        
   691→   691→        response = requests.post(self.chroma_url, json=payload)
   692→   692→        result = response.json()
   693→   693→        
   694→   694→        if result.get("audio"):
   695→   695→            return base64.b64decode(result["audio"])
   696→   696→        else:
   697→   697→            # Fallback to cascaded pipeline
   698→   698→            stt_result = await self.speech_to_text(audio_bytes, language)
   699→   699→            response_text = await self._generate_response(stt_result["text"], character)
   700→   700→            return await self.text_to_speech(response_text, language)
   701→   701→    
   702→   702→    def _get_character_prompt(self, character: str) -> str:
   703→   703→        """Get system prompt for a character."""
   704→   704→        prompts = {
   705→   705→            "kwe": "You are Kwe, a wise owl guide for Cameroonian children. Speak warmly and encouragingly.",
   706→   706→            "mbi": "You are Mbi, a curious monkey. Speak playfully and energetically.",
   707→   707→            "ngo": "You are Ngo, a brave girl from the Grassfields. Speak confidently and clearly.",
   708→   708→            "kong": "You are Kong, a resourceful boy from the coast. Speak adventurously and friendly."
   709→   709→        }
   710→   710→        return prompts.get(character, prompts["kwe"])
   711→   711→```
