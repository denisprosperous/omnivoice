   204→// ============================================================================
   205→// v4.0 VOICE STACK HEALTH + SUCCESS METRICS (Master Prompt v4.0 §I–II, §VI)
   206→// Natural Voice Stack status (Kokoro-82M / Faster-Whisper / Chroma-1.0 /
   207→// HuggingFace speech-to-speech contract) + live v4.0 success metrics:
   208→// TTS latency < 500ms, STS response < 2s, DIY completion > 70%, voice

   211→// ============================================================================
   212→interface VoiceHealth {
   213→  stack: Record<string, { stipulated: string; replaced?: string; live: boolean; fallback?: string; reachable?: boolean; vad?: string; contract?: string }>;
   214→  latency: Record<string, { n: number; p50: number | null; p95: number | null }>;
   215→  character_voices: Record<string, { voice: string; speed: number; note: string }>;
   216→  targets: Record<string, number>;
   217→}
   218→
   219→function VoiceStackPanel({ lang }: { lang: Lang }) {
   220→  const fr = lang === "fr";
   221→  const [health, setHealth] = React.useState<VoiceHealth | null>(null);
   222→  const [metrics, setMetrics] = React.useState<{
   223→    byType: Record<string, number>; feedbackAvg: number | null; feedbackN: number;
   224→  } | null>(null);
   225→
   226→  React.useEffect(() => {
   227→    fetch("/api/voice-health").then((r) => r.json()).then(setHealth).catch(() => {});
   228→    Promise.all([
   229→      fetch("/api/analytics").then((r) => r.json()).catch(() => null),
   230→      fetch("/api/feedback").then((r) => r.json()).catch(() => null),
   231→    ]).then(([analytics, feedback]) => {
   232→      const fb = Array.isArray(feedback?.submissions) ? feedback.submissions : [];
   233→      const avg = fb.length ? fb.reduce((a: number, s: { rating: number }) => a + s.rating, 0) / fb.length : null;
   234→      setMetrics({ byType: analytics?.byType || {}, feedbackAvg: avg, feedbackN: fb.length });
   235→    });
   236→  }, []);
   237→
   238→  if (!health) return null;
   239→  const stackRows = [
   240→    { key: "tts", label: "TTS", icon: "🗣️" },
   241→    { key: "stt", label: "STT", icon: "👂🏾" },
   242→    { key: "sts", label: "STS", icon: "🎭" },
   243→    { key: "pipeline", label: "Pipeline", icon: "🔗" },
   244→  ];
   245→  const lat = (k: string) => {
   246→    const l = health.latency[k];
   247→    return l && l.p95 !== null ? `${l.p95} ms (p95 · n=${l.n})` : fr ? "pas encore de mesures" : "no samples yet";
   248→  };
   249→  const byType = metrics?.byType || {};
   250→  const lessonCompletions = byType["lesson_completion"] || 0;
   251→  const diyCompletions = byType["diy_completion"] || 0;
   252→  const vpTurns = byType["voice_practice_turn"] || 0;
   253→  const vpCompletions = byType["voice_practice_completion"] || 0;
   254→  const extComplete = byType["extended_lesson_complete"] || 0;
   255→  const pct = (num: number, den: number) => (den > 0 ? Math.round((num / den) * 100) : null);
   256→
   257→  const v4Metrics: Array<{ m: string; target: string; live: string; pass: boolean | null }> = [
   258→    {
   259→      m: fr ? "Naturalité de la voix" : "Voice Naturalness", target: "4.5/5",
   260→      live: metrics?.feedbackAvg != null ? `${metrics.feedbackAvg.toFixed(1)}/5 (n=${metrics.feedbackN})` : fr ? "en attente d'avis" : "awaiting feedback",
   261→      pass: metrics?.feedbackAvg != null ? metrics.feedbackAvg >= 4.5 : null,
   262→    },
   263→    { m: fr ? "Latence TTS" : "TTS Latency", target: "< 500 ms", live: lat("tts"), pass: health.latency.tts?.p95 != null ? health.latency.tts.p95 <= 500 : null },
   264→    { m: fr ? "Latence STS" : "STS Response Time", target: "< 2 s", live: lat("sts"), pass: health.latency.sts?.p95 != null ? health.latency.sts.p95 <= 2000 : null },
   265→    { m: fr ? "Précision STT" : "STT Accuracy", target: "> 90%", live: fr ? "test de transcription requis" : "transcription testing required", pass: null },
   266→    {
   267→      m: fr ? "Taux d'achèvement DIY" : "DIY Completion Rate", target: "> 70%",
   268→      live: pct(diyCompletions, lessonCompletions) != null ? `${pct(diyCompletions, lessonCompletions)}% (${diyCompletions}/${lessonCompletions})` : fr ? "aucune donnée" : "no data",
   269→      pass: pct(diyCompletions, lessonCompletions) != null ? (pct(diyCompletions, lessonCompletions) as number) >= 70 : null,
   270→    },
   271→    {
   272→      m: fr ? "Engagement pratique vocale" : "Voice Practice Engagement", target: "> 80%",
   273→      live: vpTurns > 0 ? `${vpTurns} ${fr ? "tours" : "turns"} · ${vpCompletions} ${fr ? "terminés" : "completed"}` : fr ? "aucune donnée" : "no data",
   274→      pass: vpTurns > 0 ? vpCompletions > 0 : null,
   275→    },
   276→    {
   277→      m: fr ? "Adoption des leçons étendues" : "Lesson Extension Adoption", target: "> 60%",
   278→      live: pct(extComplete, lessonCompletions) != null ? `${pct(extComplete, lessonCompletions)}% (${extComplete}/${lessonCompletions})` : fr ? "aucune donnée" : "no data",
   279→      pass: pct(extComplete, lessonCompletions) != null ? (pct(extComplete, lessonCompletions) as number) >= 60 : null,
   280→    },
   281→    {
   282→      m: fr ? "Satisfaction des apprenants" : "Learner Satisfaction", target: "4.5/5",
   283→      live: metrics?.feedbackAvg != null ? `${metrics.feedbackAvg.toFixed(1)}/5` : fr ? "en attente d'avis" : "awaiting feedback",
   284→      pass: metrics?.feedbackAvg != null ? metrics.feedbackAvg >= 4.5 : null,
   285→    },
   286→  ];
   287→
   288→  return (
   289→    <>
   290→      <section className="rounded-2xl border-2 border-teal-300 bg-gradient-to-br from-teal-50 to-white p-4 shadow-sm">
   291→        <h3 className="mb-1 text-sm font-extrabold text-teal-900">🎙️ {t("voiceStack", lang)}</h3>
   292→        <p className="mb-3 text-[11px] text-teal-700">
   293→          {fr
   294→            ? "Stack open-source : Kokoro-82M (TTS) · Faster-Whisper (STT) · Chroma-1.0 (STS) · contrat HuggingFace speech-to-speech. Repli gracieux garanti."
   295→            : "Open-source stack: Kokoro-82M (TTS) · Faster-Whisper (STT) · Chroma-1.0 (STS) · HuggingFace speech-to-speech contract. Graceful fallback guaranteed."}
   296→        </p>
   297→        <div className="grid gap-2 sm:grid-cols-2">
   298→          {stackRows.map(({ key, label, icon }) => {
   299→            const s = health.stack[key];
   300→            if (!s) return null;
   301→            return (
   302→              <div key={key} className="rounded-xl border border-teal-100 bg-white p-2.5 text-xs">
   303→                <div className="flex items-center justify-between">
   304→                  <b className="text-teal-900">{icon} {label}: {s.stipulated}</b>
   305→                  <span className={cn(
   306→                    "rounded-full px-2 py-0.5 text-[9px] font-extrabold",
   307→                    key === "pipeline" ? (s.reachable ? "bg-lime-100 text-lime-800" : "bg-amber-100 text-amber-800") : s.live ? "bg-lime-100 text-lime-800" : "bg-amber-100 text-amber-800"
   308→                  )}>
   309→                    {key === "pipeline" ? (s.reachable ? (fr ? "EN LIGNE" : "REACHABLE") : fr ? "HORS LIGNE" : "OFFLINE") : s.live ? (fr ? "ACTIF" : "LIVE") : fr ? "REPLI" : "FALLBACK"}
   310→                  </span>
   311→                </div>
   312→                {s.fallback && <p className="mt-1 text-[10px] text-teal-600">↩ {s.fallback}</p>}
   313→              </div>
   314→            );
   315→          })}
   316→        </div>
   317→        <div className="mt-2 rounded-xl bg-white p-2.5">
   318→          <p className="text-[10px] font-bold uppercase tracking-wide text-teal-600">{fr ? "Voix Kokoro des personnages" : "Character Kokoro voices"}</p>
   319→          <ul className="mt-1 grid gap-1 text-[11px] text-teal-800 sm:grid-cols-2">
   320→            {Object.entries(health.character_voices || {}).map(([id, v]) => (
   321→              <li key={id}>🎭 <b>{id}</b> → {v.voice} ({v.speed}×)</li>
   322→            ))}
   323→          </ul>
   324→        </div>
   325→      </section>
   326→
   327→      <section className="rounded-2xl border-2 border-lime-200 bg-white p-4 shadow-sm">
   328→        <h3 className="mb-2 text-sm font-extrabold text-lime-900">🎯 {fr ? "Indicateurs de réussite v4.0 (mesure en direct)" : "v4.0 Success Metrics (live measurement)"}</h3>
   329→        <div className="overflow-x-auto rounded-xl border border-lime-100">
   330→          <table className="w-full min-w-[520px] text-left text-xs">
   331→            <thead className="bg-lime-50">
   332→              <tr className="text-[10px] uppercase text-lime-600">
   333→                <th className="px-2 py-1.5">{fr ? "Indicateur" : "Metric"}</th>
   334→                <th className="px-2 py-1.5">{fr ? "Cible" : "Target"}</th>
   335→                <th className="px-2 py-1.5">{fr ? "Mesure actuelle" : "Live"}</th>

   337→              </tr>
   338→            </thead>
   339→            <tbody>
   340→              {v4Metrics.map((row) => (
